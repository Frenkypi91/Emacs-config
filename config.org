** Auto update

#+begin_src emacs-lisp
(defvar funpkg-update-stamp-file
  (expand-file-name ".last-package-update" user-emacs-directory))
(defvar funpkg-update-log-file
  (expand-file-name "tmp/package-update.log" user-emacs-directory))

(defun funpkg--today () (format-time-string "%Y-%m-%d"))

(defun funpkg--read-stamp ()
  (when (file-exists-p funpkg-update-stamp-file)
    (with-temp-buffer
      (insert-file-contents funpkg-update-stamp-file)
      (string-trim (buffer-string)))))

(defun funpkg--write-stamp (s)
  (make-directory (file-name-directory funpkg-update-stamp-file) t)
  (with-temp-file funpkg-update-stamp-file (insert s "\n")))

(defun funpkg--batch-elisp ()
  (format
   (concat "(progn\n"
           "  (setq user-emacs-directory %S)\n"
           "  (setq package-user-dir %S)\n"
           "  (require 'package)\n"
           "  (setq package-archive-priorities '((\"melpa-stable\" . 2) (\"melpa\" . 1) (\"gnu\" . 0)))\n"
           "  (setq package-archives '((\"melpa-stable\" . \"https://stable.melpa.org/packages/\")\n"
           "                           (\"melpa\" . \"https://melpa.org/packages/\")\n"
           "                           (\"gnu\"   . \"https://elpa.gnu.org/packages/\")))\n"
           "  (package-initialize)\n"
           "  (setq package-check-signature nil)\n"
           "  (package-refresh-contents)\n"
           "  (if (fboundp 'package-upgrade-all)\n"
           "      (package-upgrade-all)\n"
           "    (dolist (pkg (mapcar #'car package-alist))\n"
           "      (ignore-errors (package-install pkg))))\n"
           "  (kill-emacs 0))\n")
   (expand-file-name user-emacs-directory)
   (expand-file-name package-user-dir)))

(defun funpkg-update-once-per-day ()
  (let ((today (funpkg--today)))
    (unless (string= today (funpkg--read-stamp))
      (funpkg--write-stamp today)
      (make-directory (file-name-directory funpkg-update-log-file) t)
      (ignore-errors (delete-file funpkg-update-log-file))
      (ignore
       (make-process
        :name     "fun-package-update"
        :noquery  t
        :command  (list (or (executable-find "emacs") "emacs")
                        "--batch" "-Q" "--eval" (funpkg--batch-elisp))
        :buffer   (find-file-noselect funpkg-update-log-file)
        :sentinel (lambda (p _)
                    (when (memq (process-status p) '(exit signal))
                      (let ((code (process-exit-status p)))
                        (unless (= code 0)
                          (message "Package update failed %d – see %s"
                                   code funpkg-update-log-file))))))))))

(add-hook 'emacs-startup-hook #'funpkg-update-once-per-day)
#+end_src

(defun funmonitor-workarea (frame)
  (or (and (fboundp 'frame-monitor-workarea) (frame-monitor-workarea frame))
      (when-let* ((attrs (and (fboundp 'frame-monitor-attributes)
                              (frame-monitor-attributes frame))))
        (alist-get 'workarea attrs))))

(defun funcenter-frame (frame)
  (when (and (frame-live-p frame) (display-graphic-p frame))
    (when-let* ((wa (funmonitor-workarea frame))
                (_ (>= (length wa) 4)))
      (let* ((mx (nth 0 wa)) (my (nth 1 wa))
             (mw (nth 2 wa)) (mh (nth 3 wa))
             (tw (max 900 (- mw funframe-margin-left funframe-margin-right)))
             (th (max 600 (- mh funframe-margin-top  funframe-margin-bottom))))
        (set-frame-size     frame tw th t)
        (set-frame-position frame (+ mx funframe-margin-left)
                                  (+ my funframe-margin-top))))))

(add-hook 'emacs-startup-hook
          (lambda () (when (display-graphic-p) (funcenter-frame (selected-frame)))))
(add-hook 'after-make-frame-functions
          (lambda (f) (with-selected-frame f (funcenter-frame f))))

            (lambda () (funcenter-frame (selected-frame)))))

#+begin_src emacs-lisp
(defvar fun/hide-dotfiles t "Hide dotfiles in file completion")

(defvar fun/dotfile--prefix-cache nil
  "Cached prefix for current minibuffer session")

(defun fun/dotfile-p (file)
  (let ((name (file-name-nondirectory (directory-file-name file))))
    (cond
     ((member name '("." "..")) nil)
     ((not fun/hide-dotfiles)   t)
     ;; Use cache instead of reading minibuffer
     ((and fun/dotfile--prefix-cache
           (string-prefix-p "." fun/dotfile--prefix-cache)) t)
     (t (not (string-prefix-p "." name))))))

(defun fun/dotfile--update-cache ()
  "Update prefix cache via post-command-hook"
  (setq fun/dotfile--prefix-cache
        (file-name-nondirectory
         (directory-file-name (minibuffer-contents-no-properties)))))

(defun fun/read-file-name (prompt &optional dir default mustmatch init pred)
  (let ((read-file-name-function nil)
        (fun/dotfile--prefix-cache nil) ; reset per session
        (combined (if pred
                      (lambda (f) (and (fun/dotfile-p f) (funcall pred f)))
                    #'fun/dotfile-p)))
    (minibuffer-with-setup-hook
        (lambda ()
          (add-hook 'post-command-hook #'fun/dotfile--update-cache nil t))
      (read-file-name prompt dir default mustmatch init combined))))

(setq read-file-name-function #'fun/read-file-name)

(defun fun/vertico-refresh ()
  (when (minibufferp)
    (setq vertico--input nil)
    (setq vertico--index -1)
    (when (boundp 'vertico--start) (setq vertico--start 0))
    (when (fboundp 'vertico--exhibit) (vertico--exhibit))))

(defun fun/toggle-dotfiles ()
  (interactive)
  (setq fun/hide-dotfiles (not fun/hide-dotfiles))
  (setq fun/dotfile--prefix-cache nil) ; invalidate cache on toggle
  (when (minibufferp)
    (minibuffer-message "Dotfiles %s" (if fun/hide-dotfiles "HIDDEN" "VISIBLE")))
  (fun/vertico-refresh))

(with-eval-after-load 'vertico
